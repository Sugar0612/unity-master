A4-R:
僵尸方法的症状是，当一个Monobehaviour的一个方法绑定了一个静态的Event，然后这个对象自己销毁了，这个静态变量的Event上绑定的方法并没有销毁，甚至可以正常调用，但是调用的数值不会改变比如一些int，string变量会一直显示最后销毁的值，但是方法里面有关于场景的比如transform.position这些的时候会报错miss，而报空是直接在引擎中报错。

public static class EventBus<T> where T : struct
{
    private static Dictionary<Type, List<Delegate>> _routes = new Dictionary<Type, List<Delegate>>();

    /// <summary>
    /// 订阅
    /// </summary>
    /// <param name="eventType"></param>
    /// <param name="action"></param>
    public static void Subscribe(Action<T> handler) 
    {
        if (!_routes.TryGetValue(typeof(T), out var list))
        {
            list = new List<Delegate>();
            _routes[typeof(T)] = list;
        }

        if (!list.Contains(handler))
            list.Add(handler);
    }

    public static void UnSubscribe(Action<T> handler)
    {
        if (!_routes.TryGetValue(typeof(T), out var list))
            return;

        if (list.Contains(handler))
        {
            list.Remove(handler);
            if (list.Count == 0)
                _routes.Remove(typeof(T));
        }
    }

    public static void Publish(in T evn)
    {
        if (!_routes.TryGetValue(typeof(T), out var list))
            return;

        for (int i = 0; i < list.Count; ++i)
        {
            ((Action<T>)list[i]).Invoke(evn);
        }
    }

    public static void ClearAll() => _routes.Clear();

    public static int RouteCount => _routes.Count;
}


public class TestScript : MonoBehaviour
{
    private PlayerDied died = new PlayerDied();

    private MonsterDied monsterDied = new MonsterDied();

    private void OnEnable()
    {
        died.Id = 12;

        EventBus<PlayerDied>.Subscribe(ThePlayerIsDied);
        EventBus<PlayerDied>.Publish(died);

        EventBus<MonsterDied>.Subscribe(TheMonsterDied);
    }

    private void ThePlayerIsDied(PlayerDied player)
    {
        UnityEngine.Debug.Log($"player id: {player.Id} is dieth.");
    }

    private void TheMonsterDied(MonsterDied monster)
    {
        UnityEngine.Debug.Log($"monster id: {monster.id} is dieth.");
    }

    private void OnDisable()
    {
        EventBus<PlayerDied>.UnSubscribe(ThePlayerIsDied);
        EventBus<MonsterDied>.UnSubscribe(TheMonsterDied);
    }
}
