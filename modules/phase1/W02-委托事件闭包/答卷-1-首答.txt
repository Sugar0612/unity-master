public static class EventBus<T, S> where T : struct
{
    public delegate void OnEvent(S arg);

    private static event OnEvent _event;

    /// <summary>
    /// 订阅
    /// </summary>
    /// <param name="eventType"></param>
    /// <param name="action"></param>
    public static void Subscribe(OnEvent event_) { _event += event_; }

    public static void UnSubscribe(OnEvent event_) { _event -= event_; }

    public static void Publish(S arg) { _event.Invoke(arg); }
}

public static class EventBus<T, S, S1> where T : struct
{
    public delegate void OnEvent(S arg, S1 arg_1);

    private static event OnEvent _event;

    /// <summary>
    /// 订阅
    /// </summary>
    /// <param name="eventType"></param>
    /// <param name="action"></param>
    public static void Subscribe(OnEvent event_) { _event += event_; }

    public static void UnSubscribe(OnEvent event_) { _event -= event_; }

    public static void Publish(S arg, S1 arg_1) { _event.Invoke(arg, arg_1); }
}

    {
        GC.Collect();

        long sink = 0, x = 10;
