public static class EventBus<T> where T : struct
{
    private static Dictionary<Type, List<Delegate>> _routes = new Dictionary<Type, List<Delegate>>();

    /// <summary>
    /// 订阅
    /// </summary>
    /// <param name="eventType"></param>
    /// <param name="action"></param>
    public static void Subscribe(Action<T> handler) 
    {
        if (!_routes.TryGetValue(typeof(T), out var list))
        {
            list = new List<Delegate>();
            _routes[typeof(T)] = list;
        }

        if (!list.Contains(handler))
            list.Add(handler);
    }

    public static void UnSubscribe(Action<T> handler)
    {
        if (!_routes.TryGetValue(typeof(T), out var list))
            return;

        if (list.Contains(handler))
        {
            list.Remove(handler);
            if (list.Count == 0)
                _routes.Remove(typeof(T));
        }
    }

    public static void Publish(in T evn)
    {
        if (!_routes.TryGetValue(typeof(T), out var list))
            return;

        for (int i = 0; i < list.Count; ++i)
        {
            ((Action<T>)list[i]).Invoke(evn);
        }
    }

    public static void ClearAll() => _routes.Clear();

    public static int RouteCount => _routes.Count;
}
